// Intersection Observer Hook for elegant scroll reveals
function useScrollReveal() {
  const ref = useRef(null);
  const [isVisible, setIsVisible] = useState(false);

  useEffect(() => {
    const observer = new IntersectionObserver(
      ([entry]) => {
        if (entry.isIntersecting) {
          setIsVisible(true);
          observer.unobserve(entry.target);
        }
      },
      { threshold: 0.1, rootMargin: '0px 0px -50px 0px' }
    );
    if (ref.current) {
      observer.observe(ref.current);
    }
    return () => observer.disconnect();
  }, []);

  return [ref, isVisible];
}

// Custom simple router
function useRouter() {
  const [currentPath, setCurrentPath] = useState('/');
  
  useEffect(() => {
    const handleHashChange = () => {
      const hash = window.location.hash.replace('#', '') || '/';
      setCurrentPath(hash);
      window.scrollTo({ top: 0, behavior: 'smooth' });
    };
    window.addEventListener('hashchange', handleHashChange);
    return () => window.removeEventListener('hashchange', handleHashChange);
  }, []);

  const navigate = (path) => {
    window.location.hash = path;
  };

  return { currentPath, navigate };
}

const RevealSection = ({ children, className = "", delay = 0 }) => {
  const [ref, isVisible] = useScrollReveal();
  return (
    <div
      ref={ref}
      className={`transition-all duration-1000 ease-out ${
        isVisible ? 'opacity-100 translate-y-0' : 'opacity-0 translate-y-12'
      } ${className}`}
      style={{ transitionDelay: `${delay}ms` }}
    >
      {children}
    </div>
  );
};

const BookingWizard = ({ lang, navigate }) => {
  const [step, setStep] = useState(1);
  const [formData, setFormData] = useState({ name: '', age: '', phone: '', service: '', complaint: '' });
  const [errors, setErrors] = useState({});
  const [isSubmitting, setIsSubmitting] = useState(false);

  const validate = () => {
    const newErrors = {};
    if (step === 1 && !formData.name) newErrors.name = true;
    if (step === 2 && !formData.age) newErrors.age = true;
    if (step === 3 && (!formData.phone || formData.phone.length < 8)) newErrors.phone = true;
    if (step === 4 && !formData.complaint) newErrors.complaint = true;
    setErrors(newErrors);
    return Object.keys(newErrors).length === 0;
  };

  const nextStep = () => {
    if (validate()) setStep(prev => prev + 1);
  };

  const submit = () => {
    setIsSubmitting(true);
    setTimeout(() => {
      const url = createWhatsAppBookingUrl(formData);
      window.open(url, '_blank');
      navigate('/');
    }, 1500);
  };

  const questionsAr = ["ما هو اسمك؟", "كم عمرك؟", "رقم هاتفك للتواصل؟", "مما تشتكي؟", "هل هناك خدمة معينة تبحث عنها؟ (اختياري)"];
  const questionsEn = ["What is your name?", "How old are you?", "Your phone number?", "What is your main complaint?", "Any specific service? (Optional)"];
  const questions = lang === 'ar' ? questionsAr : questionsEn;

  return (
    <div className="min-h-screen bg-slate-50 pt-32 pb-20 flex flex-col items-center">
      <div className="w-full max-w-2xl bg-white rounded-3xl shadow-xl p-8 md:p-12 border border-slate-100">
        
        {/* Progress */}
        {!isSubmitting && (
          <div className="flex justify-between mb-12 relative">
            <div className="absolute top-1/2 left-0 right-0 h-1 bg-slate-100 -z-10 -translate-y-1/2"></div>
            <div className="absolute top-1/2 left-0 h-1 bg-teal-500 -z-10 -translate-y-1/2 transition-all duration-500" style={{ width: `${((step - 1) / 4) * 100}%` }}></div>
            
            {[1, 2, 3, 4, 5].map(i => (
              <div key={i} className={`w-10 h-10 rounded-full flex items-center justify-center font-bold text-sm transition-colors duration-500 ${step >= i ? 'bg-teal-500 text-white shadow-lg shadow-teal-500/40' : 'bg-slate-200 text-slate-500'}`}>
                {0}{i}
              </div>
            ))}
          </div>
        )}

        {isSubmitting ? (
          <div className="text-center py-12 animate-fade-in">
            <div className="w-24 h-24 bg-teal-100 rounded-full flex items-center justify-center mx-auto mb-6">
              <CheckCircle2 size={48} className="text-teal-500" />
            </div>
            <h2 className="text-2xl font-bold text-slate-900 mb-2">
              {lang === 'ar' ? 'تم تجهيز طلبك ✓' : 'Request Prepared ✓'}
            </h2>
            <p className="text-slate-600">
              {lang === 'ar' ? 'سيتم فتح واتساب لإرسال التفاصيل إلى العيادة.' : 'WhatsApp will open to send details to the clinic.'}
            </p>
          </div>
        ) : (
          <div className="animate-fade-in" key={step}>
            <h2 className="text-2xl md:text-3xl font-black text-slate-900 mb-8 text-center">
              {questions[step - 1]}
            </h2>

            <div className="mb-8">
              {step === 1 && (
                <input 
                  type="text" 
                  value={formData.name} 
                  onChange={e => setFormData({...formData, name: e.target.value})}
                  className={`w-full text-xl p-4 rounded-xl border-2 focus:outline-none transition-colors ${errors.name ? 'border-orange-500 bg-orange-50' : 'border-slate-200 focus:border-teal-500'}`}
                  placeholder={lang === 'ar' ? 'الاسم بالكامل' : 'Full Name'}
                />
              )}
              {step === 2 && (
                <input 
                  type="number" 
                  value={formData.age} 
                  onChange={e => setFormData({...formData, age: e.target.value})}
                  className={`w-full text-xl p-4 rounded-xl border-2 focus:outline-none transition-colors ${errors.age ? 'border-orange-500 bg-orange-50' : 'border-slate-200 focus:border-teal-500'}`}
                  placeholder={lang === 'ar' ? 'العمر' : 'Age'}
                />
              )}
              {step === 3 && (
                <input 
                  type="tel" 
                  value={formData.phone} 
                  onChange={e => setFormData({...formData, phone: e.target.value})}
                  className={`w-full text-xl p-4 rounded-xl border-2 focus:outline-none transition-colors ${errors.phone ? 'border-orange-500 bg-orange-50' : 'border-slate-200 focus:border-teal-500'}`}
                  placeholder="01xxxxxxxxx"

              )}

              )}
              {step === 5 && (
                <select 
                  value={formData.service} 
                  onChange={e => setFormData({...formData, service: e.target.value})}
                  className="w-full text-xl p-4 rounded-xl border-2 border-slate-200 focus:border-teal-500 focus:outline-none bg-white"
                >
                  <option value="">{lang === 'ar' ? 'غير متأكد / فحص عام' : 'Not sure / General checkup'}</option>
                  {servicesConfig.map(s => (
                    <option key={s.slug} value={lang === 'ar' ? s.titleAr : s.titleEn}>
                      {lang === 'ar' ? s.titleAr : s.titleEn}
                    </option>
                  ))}
                </select>
              )}
            </div>

            <div className="flex justify-between">
              {step > 1 ? (
                <button 
                  onClick={() => setStep(step - 1)}
                  className="px-6 py-3 text-slate-500 font-bold hover:text-slate-800 transition-colors"
                >
                  {lang === 'ar' ? 'رجوع' : 'Back'}
                </button>
              ) : <div></div>}
              
              {step < 5 ? (
                <ButtonPrimary onClick={nextStep}>
                  {lang === 'ar' ? 'التالي' : 'Next'}
                </ButtonPrimary>
              ) : (
                <ButtonPrimary onClick={submit} className="bg-teal-600 hover:bg-teal-700 shadow-teal-600/30">
                  {lang === 'ar' ? 'إرسال عبر واتساب' : 'Send via WhatsApp'}
                  <MessageCircle size={20} className="ml-2" />
                </ButtonPrimary>
              )}
            </div>
          </div>
        )}
      </div>
    </div>
  );
};

export default function App() {
  const [lang, setLang] = useState('ar');
  const { currentPath, navigate } = useRouter();

  // Apply RTL/LTR to body
  useEffect(() => {
    document.documentElement.dir = lang === 'ar' ? 'rtl' : 'ltr';
    document.documentElement.lang = lang;
  }, [lang]);

  // View Controller
  const renderView = () => {
    if (currentPath === '/') {
      return (
        <main>
          <HeroSection lang={lang} navigate={navigate} />
          <EducationalSections lang={lang} navigate={navigate} />
          <ServicesGrid lang={lang} navigate={navigate} />
          <SmileJourney lang={lang} />
          <TrustCenter lang={lang} />
          <FAQSection lang={lang} />
        </main>
      );
    }
    if (currentPath === '/booking') {
      return <BookingWizard lang={lang} navigate={navigate} />;
    }
    if (currentPath.startsWith('/services/')) {
      const slug = currentPath.split('/')[2];
      return <ServicePage slug={slug} lang={lang} navigate={navigate} />;
    }

    // 404 Fallback
    return (
      <div className="min-h-screen flex flex-col items-center justify-center pt-20 text-center px-4">
        <div className="text-teal-200 mb-6">
          <Smile size={100} />
        </div>

          {DICTIONARY[lang].notFound} 🦷
        </h1>
        <ButtonPrimary onClick={() => navigate('/')}>
          {DICTIONARY[lang].backToHome}
        </ButtonPrimary>
      </div>
    );
  };

  return (
    <div className="font-sans text-slate-900 bg-white min-h-screen selection:bg-teal-200">
      <style dangerouslySetInnerHTML={{__html: `

        }
        
        .animate-fade-in {

        }
